-- =========================
-- DB SUPABASE SGS
-- =========================
-- =========================
-- ENUMS
-- =========================

CREATE TYPE role_enum AS ENUM ('Administrador', 'Encargado', 'Colaborador', 'Cliente');

CREATE TYPE estado_servicio_enum AS ENUM ('Pendiente', 'En progreso', 'Completado', 'Bloqueado');

CREATE TYPE tipo_nota_enum AS ENUM ('instruccion', 'comentario', 'observacion');

CREATE TYPE tipo_anuncio_enum AS ENUM ('global', 'area');

CREATE TYPE tipo_solicitud_enum AS ENUM ('apoyo', 'herramienta', 'instruccion');

CREATE TYPE estado_solicitud_enum AS ENUM ('pendiente', 'atendido');

-- ENUM para estado del servicio
CREATE TYPE estado_servicio_enum AS ENUM ('Pendiente', 'En Proceso', 'Completado', 'Cancelado');

-- TABLA CLIENTES
CREATE TABLE Clientes (
    Cliente_Id SERIAL PRIMARY KEY,
    Cliente_Dni VARCHAR(15) UNIQUE NOT NULL,
    Cliente_Apellido_Paterno VARCHAR(100),
    Cliente_Apellido_Materno VARCHAR(100),
    Cliente_Nombres VARCHAR(150),
    Cliente_Telefono VARCHAR(20)
);

-- TABLA USUARIOS
CREATE TABLE Usuarios (
    Usuario_Id SERIAL PRIMARY KEY,
    Usuario_Dni VARCHAR(15) UNIQUE,
    Usuario_Nombres VARCHAR(150),
    Usuario_Apellido_Paterno VARCHAR(100),
    Usuario_Apellido_Materno VARCHAR(100),
    Usuario_Telefono VARCHAR(20),
    Usuario_Correo VARCHAR(150) UNIQUE,
    Usuario_Rol VARCHAR(50),
    Usuario_Username VARCHAR(100) UNIQUE,
    Usuario_Contrasena TEXT,
    Usuario_Activo BOOLEAN DEFAULT TRUE,
    Usuario_Fecha_Creacion DATE DEFAULT CURRENT_DATE,
    Usuario_Hora_Creacion TIME DEFAULT CURRENT_TIME,
    Usuario_Disponible BOOLEAN DEFAULT TRUE
);

-- TABLA AREAS
CREATE TABLE Areas (
    Area_Id SERIAL PRIMARY KEY,
    Area_Nombre VARCHAR(100),
    Area_Descripcion TEXT,
    Area_Encargado_Id INT,
    CONSTRAINT fk_area_encargado FOREIGN KEY (Area_Encargado_Id)
        REFERENCES Usuarios(Usuario_Id)
);

-- TABLA SERVICIOS
CREATE TABLE Servicios (
    Servicio_Id SERIAL PRIMARY KEY,
    Servicio_Cliente_Id VARCHAR(15),
    Servicio_Area_Id INT,
    Servicio_Usuario_Id INT,
    Servicio_Nombre VARCHAR(150),
    Servicio_Descripcion_Equipo TEXT,
    Servicio_Serie_Equipo VARCHAR(100),
    Servicio_Detalles_Equipo TEXT,
    Servicio_Descripcion TEXT,
    Servicio_Fecha_Inicio DATE,
    Servicio_Hora_Inicio TIME,
    Servicio_Fecha_Fin DATE,
    Servicio_Hora_Fin TIME,
    Servicio_Estado estado_servicio_enum DEFAULT 'Pendiente',
    Servicio_Progreso INTEGER DEFAULT 0,

    CONSTRAINT fk_servicio_cliente FOREIGN KEY (Servicio_Cliente_Id)
        REFERENCES Clientes(Cliente_Dni),

    CONSTRAINT fk_servicio_area FOREIGN KEY (Servicio_Area_Id)
        REFERENCES Areas(Area_Id),

    CONSTRAINT fk_servicio_usuario FOREIGN KEY (Servicio_Usuario_Id)
        REFERENCES Usuarios(Usuario_Id)
);






-- =========================
-- AREAS
-- =========================

CREATE TABLE areas (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    nombre TEXT NOT NULL,
    descripcion TEXT,
    encargado UUID
);


-- =========================
-- USUARIOS
-- =========================

CREATE TABLE usuarios (
    id_usuario UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    dni TEXT UNIQUE,
    nombres TEXT NOT NULL,
    apellido_paterno TEXT,
    apellido_materno TEXT,
    telefono TEXT,
    correo TEXT UNIQUE,
    username TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    rol role_enum NOT NULL,
    
    id_area_principal UUID REFERENCES areas(id),
    id_area_adicional UUID REFERENCES areas(id),
    
    encargado_area_principal BOOLEAN DEFAULT FALSE,
    encargado_area_adicional BOOLEAN DEFAULT FALSE,

    fecha_actualizacion_password TIMESTAMP,
    ultimo_login TIMESTAMP,
    fecha_creacion TIMESTAMP DEFAULT NOW(),
    
    activo BOOLEAN DEFAULT TRUE,
    ocupado BOOLEAN DEFAULT FALSE
);


-- =========================
-- AREAS - COLABORADORES (RELACION)
-- =========================

CREATE TABLE area_colaboradores (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    id_area UUID REFERENCES areas(id) ON DELETE CASCADE,
    id_usuario UUID REFERENCES usuarios(id_usuario) ON DELETE CASCADE
);


-- =========================
-- SERVICE TEMPLATES
-- =========================

CREATE TABLE service_templates (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    nombre TEXT NOT NULL,
    descripcion TEXT,
    area UUID REFERENCES areas(id),
    activo BOOLEAN DEFAULT TRUE,
    fecha_creacion TIMESTAMP DEFAULT NOW()
);


-- =========================
-- TEMPLATE TAREAS
-- =========================

CREATE TABLE template_tareas (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    id_template UUID REFERENCES service_templates(id) ON DELETE CASCADE,
    nombre TEXT NOT NULL,
    orden INTEGER
);


-- =========================
-- SERVICIOS
-- =========================

CREATE TABLE servicios (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    codigo TEXT UNIQUE,
    cliente TEXT,
    telefono_cliente TEXT,
    descripcion TEXT,
    area UUID REFERENCES areas(id),
    
    fecha_inicio DATE,
    hora_inicio TIME,
    
    fecha_fin DATE,
    hora_fin TIME,
    
    hora_estimada_fin TIME,
    inicio_real TIMESTAMP,

    estado estado_servicio_enum DEFAULT 'Pendiente',
    progreso INTEGER DEFAULT 0
);


-- =========================
-- TECNICOS POR SERVICIO
-- =========================

CREATE TABLE servicio_tecnicos (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    id_servicio UUID REFERENCES servicios(id) ON DELETE CASCADE,
    id_usuario UUID REFERENCES usuarios(id_usuario)
);


-- =========================
-- TAREAS
-- =========================

CREATE TABLE tareas (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    id_servicio UUID REFERENCES servicios(id) ON DELETE CASCADE,
    nombre TEXT NOT NULL,
    completada BOOLEAN DEFAULT FALSE,
    fecha_completada TIMESTAMP,
    responsable UUID REFERENCES usuarios(id_usuario),
    orden INTEGER
);


-- =========================
-- NOTAS DE TAREA
-- =========================

CREATE TABLE task_notes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    id_tarea UUID REFERENCES tareas(id) ON DELETE CASCADE,
    autor UUID REFERENCES usuarios(id_usuario),
    rol role_enum,
    texto TEXT,
    tipo tipo_nota_enum,
    fecha TIMESTAMP DEFAULT NOW()
);


-- =========================
-- COMENTARIOS
-- =========================

CREATE TABLE comentarios (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    id_servicio UUID REFERENCES servicios(id) ON DELETE CASCADE,
    autor UUID REFERENCES usuarios(id_usuario),
    rol role_enum,
    texto TEXT,
    fecha TIMESTAMP DEFAULT NOW()
);


-- =========================
-- ANUNCIOS
-- =========================

CREATE TABLE anuncios (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    titulo TEXT,
    contenido TEXT,
    autor UUID REFERENCES usuarios(id_usuario),
    fecha TIMESTAMP DEFAULT NOW(),
    tipo tipo_anuncio_enum,
    area_destino UUID REFERENCES areas(id)
);


-- =========================
-- SOLICITUDES INTERNAS
-- =========================

CREATE TABLE solicitudes_internas (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tipo tipo_solicitud_enum,
    solicitante UUID REFERENCES usuarios(id_usuario),
    destinatario UUID REFERENCES usuarios(id_usuario),
    contenido TEXT,
    fecha TIMESTAMP DEFAULT NOW(),
    estado estado_solicitud_enum DEFAULT 'pendiente'
);


-- =========================
-- AUDITORIA
-- =========================

CREATE TABLE audit_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    usuario UUID REFERENCES usuarios(id_usuario),
    accion TEXT,
    modulo TEXT,
    detalle TEXT,
    fecha TIMESTAMP DEFAULT NOW()
);
